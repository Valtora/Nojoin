# Packages installed without their declared dependencies, after worker.txt or
# any file that includes it (test.txt, dev.txt, local.txt):
#
#   python -m pip install --no-deps -r requirements/worker-no-deps.txt
#
# Never include this file from another one: pip would then resolve the
# dependencies this file exists to skip.
#

# for 20 advisories (#526). Nojoin uses fastembed only for TextEmbedding; the
# cap serves its image models, which Nojoin never loads. worker.txt pins the
# rest of what it declares, and backend/tests/test_no_deps_requirements.py
# checks those pins against fastembed's metadata, since pip no longer does.
# When a fastembed release accepts the pinned pillow, that test fails: move
# fastembed back into worker.txt and delete this file and its install lines.
